package greet

import "testing"

func TestHello(t *testing.T) {
	if got := Hello("Ada"); got != "Hello, Ada" {
		t.Errorf("Hello(%q) = %q, want %q", "Ada", got, "Hello, Ada")
	}
}

func TestHelloEmptyName(t *testing.T) {
	if got := Hello(""); got != "Hello, stranger" {
		t.Errorf("Hello(%q) = %q, want %q", "", got, "Hello, stranger")
	}
}
