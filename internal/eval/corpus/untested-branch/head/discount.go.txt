package main

// Discount returns the total price of qty units at price each, taking 15% off
// orders of 100 units or more and 5% off orders of 10 units or more.
func Discount(qty int, price float64) float64 {
	total := float64(qty) * price
	if qty >= 100 {
		return total * 0.85
	}
	if qty >= 10 {
		return total * 0.95
	}
	return total
}
